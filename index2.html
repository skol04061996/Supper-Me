/**
 * SUPPER ME KOLKATA — 2-WAY GOOGLE SHEETS ATTENDANCE & ROSTER BRIDGE
 * 1. Create a Google Sheet -> Extensions -> Apps Script -> Paste this code.
 * 2. Deploy -> New deployment -> Web app -> Execute as: Me -> Access: Anyone.
 * 3. Paste the Web App URL into the Supper Me Employer Portal.
 */

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var payload = JSON.parse(e.postData.contents);
    var type = payload.type || 'attendance';

    if (type === 'ping') {
      setupSheets(ss);
      return jsonOut({ status: 'ok', message: 'Connected to Supper Me Sheet' });
    }

    if (type === 'staff_enroll') {
      var staffSheet = getOrCreate(ss, 'Staff_Roster', [
        'Employee ID', 'Full Name', 'Role', 'Outlet', 'Shift Start', 'Hourly Rate', 'Staff PIN'
      ]);
      var s = payload.staff;
      staffSheet.appendRow([
        s.id, s.name, s.role, s.outlet, s.shiftStart, s.hourlyRate, s.pin || '0000'
      ]);
      return jsonOut({ status: 'ok', message: 'Staff enrolled' });
    }

    // Default: Attendance Punch (Clock In or Clock Out)
    var logSheet = getOrCreate(ss, 'Attendance_Logs', [
      'Log ID', 'Date', 'Employee ID', 'Name', 'Role', 'Outlet', 'Action', 'Time', 'Status', 'Duration', 'Match %'
    ]);
    var l = payload.log;
    logSheet.appendRow([
      l.id,
      l.date,
      l.employeeId,
      l.name,
      l.role,
      l.outlet,
      l.action,
      l.time,
      l.status,
      l.duration || 'In Progress',
      l.matchScore || '98.0%'
    ]);

    return jsonOut({ status: 'ok', logId: l.id });
  } catch (err) {
    return jsonOut({ status: 'error', message: err.toString() });
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    setupSheets(ss);

    var logSheet = ss.getSheetByName('Attendance_Logs');
    var logRows = logSheet.getDataRange().getValues();
    var logs = [];
    for (var i = 1; i < logRows.length; i++) {
      var r = logRows[i];
      if (!r[0]) continue;
      logs.push({
        id: String(r[0]),
        date: String(r[1]),
        employeeId: String(r[2]),
        name: String(r[3]),
        role: String(r[4]),
        outlet: String(r[5]),
        action: String(r[6]),
        time: String(r[7]),
        status: String(r[8]),
        duration: String(r[9]),
        matchScore: String(r[10])
      });
    }

    var staffSheet = ss.getSheetByName('Staff_Roster');
    var staffRows = staffSheet.getDataRange().getValues();
    var staff = [];
    for (var j = 1; j < staffRows.length; j++) {
      var sr = staffRows[j];
      if (!sr[0]) continue;
      staff.push({
        id: String(sr[0]),
        name: String(sr[1]),
        role: String(sr[2]),
        outlet: String(sr[3]),
        shiftStart: String(sr[4]),
        hourlyRate: Number(sr[5]) || 200,
        pin: String(sr[6] || '0000')
      });
    }

    return jsonOut({ status: 'ok', logs: logs, staff: staff });
  } catch (err) {
    return jsonOut({ status: 'error', message: err.toString() });
  }
}

function setupSheets(ss) {
  getOrCreate(ss, 'Attendance_Logs', [
    'Log ID', 'Date', 'Employee ID', 'Name', 'Role', 'Outlet', 'Action', 'Time', 'Status', 'Duration', 'Match %'
  ]);
  getOrCreate(ss, 'Staff_Roster', [
    'Employee ID', 'Full Name', 'Role', 'Outlet', 'Shift Start', 'Hourly Rate', 'Staff PIN'
  ]);
}

function getOrCreate(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#10b981').setFontColor('#090d16');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function jsonOut(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}
